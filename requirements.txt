# Dataset loaders (datasets/<name>/load.py)
huggingface_hub    # dataset downloads (GutenQA, Qasper, TriviaQA, TechQA, ...)
pandas             # parquet-shipped datasets
pyarrow            # parquet engine for pandas
requests           # Wikipedia / gov.uk / Gutenberg fetching
tqdm               # loader progress
chardet            # encoding detection (LiteraryQA)
beautifulsoup4     # HTML parsing
html5lib           # lenient HTML parser used by BeautifulSoup
ftfy               # mojibake repair (LiteraryQA)
loguru             # logging in the vendored LiteraryQA cleaning code
markdownify        # HTML -> Markdown renderings

# Chunking — added per method as they are implemented
pysbd               # sentence segmentation with character offsets (en, pl)
lingua-language-detector  # language="auto": which pysbd rules a document needs
llama-index-core    # node parsers: sentence, markdown; offsets are native
llama-index-embeddings-openai  # semantic chunking, any OpenAI-compatible endpoint

# Chunk augmentation
openai              # chat completions: LumberChunker, summary and contextual

# Embedding and ingestion (embedding.py, ingest.py)
chromadb                          # persistent vector database, one collection per embedding model
llama-index-vector-stores-chroma  # ChromaVectorStore over that collection
llama-index-retrievers-bm25       # BM25Retriever with disk persistence (bm25s, PyStemmer)
tokenizers                        # an embedding model's own tokenizer, for exact truncation
tiktoken                          # the stand-in tokenizer when none is named

# Evaluation (evaluate_ragas.py)
ragas                       # context precision and recall, judged by an LLM
langchain-community<0.4     # ragas 0.4.3 imports chat_models.vertexai, which 0.4 removed

# Visualization
streamlit             # interactive chunk visualizer UI
matplotlib            # charts in experiments/results_analysis.ipynb
ipykernel             # runs that notebook
